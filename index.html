function updateHistory() { const history = JSON.parse(localStorage.getItem('taxi_history') || '[]'); const mode = document.getElementById('view-mode').value; const selYear = document.getElementById('view-year').value; const selMonth = document.getElementById('view-month').value; const list = document.getElementById('history-list'); const head = document.getElementById('list-head'); let totalS = 0, totalI = 0; if (mode === 'month') { let targetYM; if (selMonth === 'all') { // 今月 const now = new Date(); targetYM = now.getFullYear() + '-' + String(now.getMonth()+1).padStart(2,'0'); } else { // 指定月 targetYM = selYear + '-' + selMonth; } document.getElementById('total-label').innerText = targetYM.replace('-', '年') + '月の合計'; head.innerHTML = '日付売上操作'; const filtered = history.filter(d => d.date.startsWith(targetYM)); filtered.forEach(d => totalS += d.sales); const r = getRate(totalS, true); totalI = Math.floor(totalS * (r / 100)); document.getElementById('total-rate-badge').innerText = totalS < 420000 ? `暫定歩合: ${r}%` : `確定歩合: ${r}%`; filtered.sort((a,b) => new Date(b.date) - new Date(a.date)); list.innerHTML = filtered.map(d => ` ${d.date.substring(5).replace('-','/')} ${d.sales.toLocaleString()}
`).join(''); } else { // 年集計（既存そのまま） document.getElementById('total-label').innerText = selYear + '年の合計'; head.innerHTML = '月売上手取り'; const yearData = {}; history.filter(d => d.date.startsWith(selYear)).forEach(d => { const m = d.date.substring(0,7); if (!yearData[m]) yearData[m] = 0; yearData[m] += d.sales; }); const months = Object.keys(yearData).sort().reverse(); list.innerHTML = months.map(m => { const r = getRate(yearData[m], true); const inc = Math.floor(yearData[m] * (r / 100)); totalS += yearData[m]; totalI += inc; return ` ${m.substring(5)}月 ${yearData[m].toLocaleString()} ${inc.toLocaleString()} `; }).join(''); document.getElementById('total-rate-badge').innerText = '年間合計'; } document.getElementById('total-sales').innerText = totalS.toLocaleString(); document.getElementById('total-income').innerText = totalI.toLocaleString() + '円'; }
